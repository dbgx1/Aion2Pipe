# Identify the actual source build in logs even when no Git checkout is present.
file(GLOB build_sources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.cpp" "${CMAKE_SOURCE_DIR}/src/*.hpp")
list(APPEND build_sources "${CMAKE_SOURCE_DIR}/CMakeLists.txt")
list(SORT build_sources)
set(build_fingerprint_input "")
foreach(build_source IN LISTS build_sources)
  file(SHA256 "${build_source}" source_hash)
  get_filename_component(source_name "${build_source}" NAME)
  string(APPEND build_fingerprint_input "${source_name}:${source_hash}\n")
endforeach()
string(SHA256 build_fingerprint "${build_fingerprint_input}")
target_compile_definitions(diagnostics PRIVATE AION_BUILD_ID="${build_fingerprint}")
target_include_directories(diagnostics PUBLIC src)

function(copy_runtime target)
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/data/skin_catalog.json" "$<TARGET_FILE_DIR:${target}>/skin_catalog.json"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WINDIVERT_DIR}/x64/WinDivert.dll" "$<TARGET_FILE_DIR:${target}>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WINDIVERT_DIR}/x64/WinDivert64.sys" "$<TARGET_FILE_DIR:${target}>")
endfunction()
